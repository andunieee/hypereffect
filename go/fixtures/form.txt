# A form with an indicator; local signals are not sent.
!start
< hello {}
=signals {}
signal {"login":{"user":"","pass":""},"error":"","_busy":false}
element {"id":"login","type":"card","title":"Sign in"}
element {"id":"user","type":"input","parent":"login","label":"Username","bind":"login.user"}
element {"id":"pass","type":"input","parent":"login","label":"Password","variant":"password","bind":"login.pass","onSubmit":{"action":"login","indicator":"_busy"}}
element {"id":"err","type":"notice","parent":"login","variant":"failure","text":{"cel":"s.error"},"visible":{"cel":"s.error != ''"}}
element {"id":"go","type":"button","parent":"login","variant":"primary","label":{"cel":"s._busy ? 'Signing in…' : 'Sign in'"},"disabled":{"cel":"s._busy || s.login.user == ''"},"onPress":{"action":"login","indicator":"_busy"}}
=attrs login {"title":"Sign in","gap":"medium"}
=attrs user {"bind":"login.user","placeholder":null,"error":"","variant":"text","onChange":null}
=attrs go {"disabled":true,"label":"Sign in"}
=attrs err {"visible":false}
# A disabled button does not run its handler.
!trigger go onPress
!write login.user "ada"
!write login.pass "hunter2"
=attrs go {"disabled":false}
!trigger go onPress
< action {"name":"login","signals":{"login":{"user":"ada","pass":"hunter2"},"error":""}}
=attrs go {"disabled":true,"label":"Signing in…"}
signal {"_busy":false,"error":"Wrong username or password.","login":{"pass":""}}
=attrs err {"visible":true,"text":"Wrong username or password."}
=attrs go {"label":"Sign in","disabled":false}
=signals {"login":{"user":"ada","pass":""},"error":"Wrong username or password.","_busy":false}
# Writing through a non-object creates the intermediate objects.
!write error.detail 1
=signals {"login":{"user":"ada","pass":""},"error":{"detail":1},"_busy":false}
